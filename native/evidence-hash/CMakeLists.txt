cmake_minimum_required(VERSION 3.16)
project(evidence_hash LANGUAGES CXX)
add_executable(evidence-hash main.cpp)
target_compile_features(evidence-hash PRIVATE cxx_std_17)
if(WIN32)
  target_link_libraries(evidence-hash PRIVATE bcrypt)
else()
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(evidence-hash PRIVATE OpenSSL::Crypto)
endif()
